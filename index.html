/* 통신 성공률을 높인 강화된 fetch 로직 */ async function fetchReliefWeb(iso, days, limit) { // 1. 보안을 위해 appname을 일반적인 테스트용으로도 시도 const testAppName = "ewha_recap_v2_fix"; const dateLimit = new Date(); dateLimit.setDate(dateLimit.getDate() - days); const dateStr = dateLimit.toISOString().split('T')[0]; // 2. 최신 API v1 엔드포인트 사용 (v2보다 안정적일 때가 있음) const url = `https://api.reliefweb.int/v1/reports?appname=${testAppName}&filter[operator]=AND&filter[conditions][0][field]=country.iso3&filter[conditions][0][value]=${iso}&filter[conditions][1][field]=date.created&filter[conditions][1][value][from]=${dateStr}T00:00:00%2B00:00&limit=${limit}&profile=full`; try { const response = await fetch(url, { method: 'GET', mode: 'cors', // CORS 명시 headers: { 'Accept': 'application/json' } }); if (!response.ok) throw new Error(`HTTP ${response.status}`); return await response.json(); } catch (e) { console.error(`${iso} 수집 실패:`, e); return null; } }
