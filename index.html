// 1. 기존 코드의 rw-fetch 클릭 이벤트를 아래와 같이 교체하세요. document.getElementById('rw-fetch').onclick = async function() { const status = document.getElementById('rw-status'); const days = document.getElementById('rw-days').value; const limit = document.getElementById('rw-limit').value; status.textContent = "데이터 수집 중..."; currentItems = []; // 초기화 // 선택된 국가가 없으면 전체 20개국 진행 const targets = pickedISO3.length ? pickedISO3 : RECAP_ISO3; for (const iso of targets) { status.textContent = `${iso} 분석 중...`; const data = await fetchReliefWeb(iso, days, limit); // 작성하신 강화된 로직 호출 if (data && data.data) { // 데이터 파싱 로직 (원본 코드의 data.data.forEach 부분 유지) data.data.forEach(item => { const f = item.fields; const fullText = (f.title + " " + (f.body || "")); // 원본 코드의 핵심: HDP, Risk, C15 분석 함수들 연결 const hdp = hdpScore(fullText); const risk = analyzeRiskAndConflict(fullText); const c15 = classifyC15(fullText, 3); currentItems.push({ id: item.id, title: f.title, country: iso, created: f.date.created, hdp: hdp, risk: risk, c15: c15, url: f.url, source: f.source[0].name }); }); } } status.textContent = "수집 완료!"; renderDashboardTiles(); // 대시보드 갱신 renderResultsTable(); // 테이블 갱신 };
